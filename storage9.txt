9.1 MainActivity.java
package com.example.myapp9;

import android.content.Context;
import android.os.Bundle;
import android.view.View;
import android.widget.EditText;
import android.widget.RadioButton;
import android.widget.Toast;

import androidx.appcompat.app.AppCompatActivity;

import java.io.*;
import java.nio.charset.StandardCharsets;

public class MainActivity extends AppCompatActivity {

    EditText dataInput, fileNameInput;
    RadioButton internalRadio, externalRadio;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        dataInput = findViewById(R.id.datatext);
        fileNameInput = findViewById(R.id.filenametext);
        internalRadio = findViewById(R.id.internalButton);
        externalRadio = findViewById(R.id.externalButton);

        Toast.makeText(this, "App Started", Toast.LENGTH_SHORT).show();
    }

    public void saveData(View view) {
        String file = fileNameInput.getText().toString();
        String data = dataInput.getText().toString();

        if (file.isEmpty() || data.isEmpty()) {
            toast("Enter filename and data");
            return;
        }

        try {
            OutputStream os;

            if (internalRadio.isChecked())
                os = openFileOutput(file, Context.MODE_PRIVATE);
            else if (externalRadio.isChecked())
                os = new FileOutputStream(new File(getExternalFilesDir(null), file));
            else {
                toast("Select Internal or External");
                return;
            }

            os.write(data.getBytes(StandardCharsets.UTF_8));
            os.close();
            toast("Saved Successfully");

        } catch (Exception e) {
            toast("Error: " + e.getMessage());
        }
    }

    public void loadData(View view) {
        String file = fileNameInput.getText().toString();

        if (file.isEmpty()) {
            toast("Enter filename");
            return;
        }

        try {
            InputStream is;

            if (internalRadio.isChecked())
                is = openFileInput(file);
            else if (externalRadio.isChecked())
                is = new FileInputStream(new File(getExternalFilesDir(null), file));
            else {
                toast("Select Internal or External");
                return;
            }

            BufferedReader br = new BufferedReader(new InputStreamReader(is));
            StringBuilder sb = new StringBuilder();
            String line;

            while ((line = br.readLine()) != null)
                sb.append(line).append("\n");

            br.close();
            dataInput.setText(sb.toString().trim());
            toast("Loaded Successfully");

        } catch (Exception e) {
            toast("File not found");
        }
    }

    void toast(String msg) {
        Toast.makeText(this, msg, Toast.LENGTH_SHORT).show();
    }
}
